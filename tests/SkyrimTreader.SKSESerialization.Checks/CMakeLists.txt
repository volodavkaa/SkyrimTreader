cmake_minimum_required(VERSION 3.24)
project(STRSKSESerializationChecks LANGUAGES CXX)
set(SKSE_SDK_ROOT "" CACHE PATH "Extracted SKSE 2.3.1 source")
if(NOT EXISTS "${SKSE_SDK_ROOT}/skse64/PluginAPI.h")
    message(FATAL_ERROR "Set SKSE_SDK_ROOT")
endif()
add_executable(STRSKSESerializationChecks main.cpp)
target_compile_features(STRSKSESerializationChecks PRIVATE cxx_std_20)
target_compile_definitions(STRSKSESerializationChecks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(STRSKSESerializationChecks PRIVATE "${SKSE_SDK_ROOT}")
